<script>
const API = '/api'
let timelineData = null
let replayData = null

async function get(path) {
  const r = await fetch(API + path)
  if (!r.ok) throw new Error(path + ' ' + r.status)
  return await r.json()
}

async function post(path, body) {
  const r = await fetch(API + path, {
    method:'POST',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify(body)
  })
  if (!r.ok) throw new Error(path + ' ' + r.status)
  return await r.json()
}

function k(id, v) { document.getElementById(id).textContent = v ?? '—' }
function j(id, data) { document.getElementById(id).textContent = JSON.stringify(data, null, 2) }

function renderTimeline() {
  const el = document.getElementById('timeline')
  el.innerHTML = ''
  const entries = timelineData?.entries || []

  for (const e of entries) {
    const div = document.createElement('div')
    div.className = 'entry'
    div.onclick = () => j('selected', e)

    const refs = (e.asset_refs || []).slice(0,4).join(' · ')
    div.innerHTML = `
      <div class="entry-head">
        <b>${e.entry_type}</b>
        <span class="pill ${e.severity || 'INFO'}">${e.severity || 'INFO'}</span>
      </div>
      <span>${e.title || ''}</span>
      <small>${e.timestamp || ''}</small>
      <em>${e.correlation_id || 'no-correlation'}</em>
      <em>${refs}</em>
    `
    el.appendChild(div)
  }
}

async function loadAll() {
  const health = await get('/health')
  timelineData = await get('/timeline/evidence?limit=250')
  replayData = await get('/timeline/replay')

  document.getElementById('badge').textContent = health.version + ' · ' + health.operational_mode
  k('k_backend', health.version)
  k('k_entries', timelineData.count)
  k('k_replay', replayData.replay_count)

  const c = timelineData.entry_type_counts || {}
  k('k_events', c.CLOUDEVENT || 0)
  k('k_rf_field', c.RF_FIELD_RUN || 0)
  k('k_cov', c.RF_COVERAGE_RUN || 0)

  renderTimeline()
  j('replay', replayData)
}

async function runReplay() {
  replayData = await get('/timeline/replay')
  j('replay', replayData)
}

async function runRfField() {
  await post('/rf-field/run', {
    mission_id:'MISSION-FULL-TELCO-BOOT-001',
    cell_asset_id:'CELL-N78-A',
    target_asset_id:'UAV-ALPHA-001',
    frequency_hz:3500000000,
    tx_power_dbm:43,
    tx_gain_dbi:18,
    antenna_max_dimension_m:0.8,
    azimuth_deg:120,
    mechanical_tilt_deg:4,
    electrical_tilt_deg:2
  })
  await loadAll()
}

async function runCoverage() {
  await post('/rf-coverage/run', {
    mission_id:'MISSION-FULL-TELCO-BOOT-001',
    cell_asset_id:'CELL-N78-A',
    target_asset_id:'UE-REMOTE-001',
    frequency_hz:3500000000,
    tx_power_dbm:43,
    tx_gain_dbi:18,
    rx_gain_dbi:0,
    bandwidth_hz:100000000,
    noise_figure_db:7,
    grid_extent_m:600,
    grid_step_m:100
  })
  await loadAll()
}

loadAll().catch(e => {
  document.getElementById('badge').textContent = 'ERROR'
  j('selected', {error:String(e)})
})
setInterval(loadAll, 20000)
