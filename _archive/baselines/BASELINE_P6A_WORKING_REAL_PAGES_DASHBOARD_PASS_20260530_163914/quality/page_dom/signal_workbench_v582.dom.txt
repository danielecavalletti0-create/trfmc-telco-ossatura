<script>
const $=id=>document.getElementById(id);
let lastSpectrum=[], lastPeaks=[], wfRows=[], selectedPeak=null;

function log(msg,obj){
  $('log').textContent=`[${new Date().toISOString()}] ${msg}\n${obj?JSON.stringify(obj,null,2).slice(0,5000):''}\n`+$('log').textContent;
}
async function api(path,opt={}){
  const r=await fetch(path,{headers:{'Content-Type':'application/json'},...opt});
  if(!r.ok) throw new Error(await r.text());
  return await r.json();
}
function hzMHz(id){return Number($(id).value)*1e6}
function drawSpectrum(points, mode='db'){
  const c=$('spectrum'),ctx=c.getContext('2d'),w=c.width,h=c.height;
  ctx.fillStyle='#02040a';ctx.fillRect(0,0,w,h);
  for(let i=0;i<=10;i++){ctx.strokeStyle=i%5?'rgba(127,178,198,.08)':'rgba(0,229,255,.18)';ctx.beginPath();ctx.moveTo(0,i*h/10);ctx.lineTo(w,i*h/10);ctx.stroke()}
  if(!points||!points.length)return;
  lastSpectrum=points;
  const vals=points.map(p=>p.db??p.dbm??-120), minDb=Math.min(...vals), maxDb=Math.max(...vals);
  const minF=points[0].freq_hz, maxF=points[points.length-1].freq_hz;
  ctx.strokeStyle='#00e5ff';ctx.lineWidth=2;ctx.beginPath();
  points.forEach((p,i)=>{const db=p.db??p.dbm??-120;const x=(p.freq_hz-minF)/(maxF-minF+1e-9)*w;const y=h-((db-minDb)/(maxDb-minDb+1e-9))*h;if(i===0)ctx.moveTo(x,y);else ctx.lineTo(x,y)});
  ctx.stroke();
  const pk=[...points].sort((a,b)=>(b.db??b.dbm)-(a.db??a.dbm))[0];
  if(pk){const f=pk.freq_hz||0; $('kPeakMHz').textContent=(f/1e6).toFixed(6);}
  addWaterfall(points);
}
function addWaterfall(points){
  const c=$('waterfall'),ctx=c.getContext('2d'),w=c.width,h=c.height;
  const row=points.map(p=>p.db??p.dbm??-120);wfRows.unshift(row);wfRows=wfRows.slice(0,h);
  const img=ctx.createImageData(w,h);
  for(let y=0;y<h;y++){const src=wfRows[y]||[];for(let x=0;x<w;x++){const idx=Math.floor(x/w*Math.max(1,src.length-1));const db=src[idx]??-120;const v=Math.max(0,Math.min(1,(db+110)/75));const o=(y*w+x)*4;img.data[o]=Math.floor(10+30*v);img.data[o+1]=Math.floor(30+220*v);img.data[o+2]=Math.floor(70+160*(1-v));img.data[o+3]=255;}}
  ctx.putImageData(img,0,0);
}
function renderMetrics(m){
  $('kClass').textContent=m.classification_hint||'—';
  $('kObw').textContent=m.occupied_bw_hz??'—';
  $('kPeak').textContent=m.peak_offset_hz??'—';
  $('kClip').textContent=m.clip_percent??'—';
  $('metrics').textContent=JSON.stringify(m,null,2);
}
function renderPeaks(peaks){
  lastPeaks=peaks||[];const tb=$('peaks');tb.innerHTML='';
  lastPeaks.forEach((p,i)=>{const tr=document.createElement('tr');tr.innerHTML=`<td>${p.offset_hz??p.freq_hz??''}</td><td>${p.db??p.dbm??''}</td><td>${p.snr_over_median_db??''}</td><td><button onclick="usePeak(${i})">USE</button></td>`;tb.appendChild(tr)});
}
function usePeak(i){
  selectedPeak=lastPeaks[i];
  const off=selectedPeak.offset_hz??0;
  $('off').value=off;
  log('Marker selezionato',selectedPeak);
}
function fineFromPeak(){
  if(lastPeaks.length){usePeak(0);}
  else log('Nessun peak: esegui prima ANALYZE PLUS o SWEEP WINDOW');
}
async function state(){
  try{const j=await api('/api/v581/selftest');$('apiBadge').textContent='API OK';$('apiBadge').className='badge ok';log('Selftest API',j)}
  catch(e){$('apiBadge').textContent='API ERROR';$('apiBadge').className='badge warn';log('API error',e.message)}
}
async function sweepWindow(){
  try{
    const body={start_hz:hzMHz('startMHz'),stop_hz:hzMHz('stopMHz'),bin_hz:+$('binHz').value,points:1200,use_hackrf:true};
    const j=await api('/api/v580/workbench/sweep/window',{method:'POST',body:JSON.stringify(body)});
    drawSpectrum(j.points||[]);
    const peaks=(j.peaks||[]).map(p=>({offset_hz:p.freq_hz,db:p.dbm,snr_over_median_db:''}));
    renderPeaks(peaks);
    log('Sweep window', {real_hackrf:j.real_hackrf, peaks:j.peaks, saved:j.saved});
  }catch(e){log('Errore sweep',e.message)}
}
async function loadFiles(){
  try{
    const j=await api('/api/v581/iq/files');const sel=$('iqFile');sel.innerHTML='';
    (j.iq||[]).forEach(f=>{const o=document.createElement('option');o.value=f.name;o.textContent=f.name;sel.appendChild(o)});
    const tb=$('files');tb.innerHTML='';
    ['iq','wav','reports'].forEach(k=>(j[k]||[]).forEach(f=>{const tr=document.createElement('tr');tr.innerHTML=`<td>${k}</td><td>${f.name}</td><td>${f.size}</td>`;tb.appendChild(tr)}));
    log('Files caricati',{iq:(j.iq||[]).length,wav:(j.wav||[]).length,reports:(j.reports||[]).length});
  }catch(e){log('Errore files',e.message)}
}
async function selftestSignal(){
  try{const j=await api('/api/v581/iq/selftest_signal',{method:'POST',body:JSON.stringify({mode:$('mode').value,sample_rate:+$('sr').value,seconds:2,tone_hz:1000,carrier_offset_hz:+$('off').value})});log('Selftest IQ creato',j);await loadFiles();}
  catch(e){log('Errore selftest',e.message)}
}
async function analyzePlus(){
  try{
    const j=await api('/api/v581/iq/analyze_plus',{method:'POST',body:JSON.stringify({filename:$('iqFile').value||null,sample_rate:+$('sr').value,fft_size:8192,max_seconds:+$('sec').value})});
    drawSpectrum(j.spectrum||[]);renderMetrics(j.metrics||{});renderPeaks(j.peaks||[]);log('Analyze Plus', {file:j.file, metrics:j.metrics, peaks:j.peaks});
  }catch(e){log('Errore analyze',e.message)}
}
async function demod(){
  try{
    const body={filename:$('iqFile').value||null,sample_rate:+$('sr').value,freq_offset_hz:+$('off').value,channel_bw_hz:+$('bw').value,mode:$('mode').value,max_seconds:+$('sec').value,audio_rate:48000};
    const j=await api('/api/v581/iq/channelize_demod',{method:'POST',body:JSON.stringify(body)});
    $('player').src=j.wav_url;$('kWav').textContent=j.wav_file||'OK';renderMetrics(j.channel_metrics||{});log('Channelize + Demod',j);await loadFiles();
  }catch(e){log('Errore demod',e.message)}
}
async function modes(){try{log('Modes',await api('/api/v581/demod/modes'))}catch(e){log('Errore modes',e.message)}}
async function lastReport(){try{log('Last report',await api('/api/v581/iq/last_report'))}catch(e){log('Errore report',e.message)}}
state();loadFiles();
</script>
