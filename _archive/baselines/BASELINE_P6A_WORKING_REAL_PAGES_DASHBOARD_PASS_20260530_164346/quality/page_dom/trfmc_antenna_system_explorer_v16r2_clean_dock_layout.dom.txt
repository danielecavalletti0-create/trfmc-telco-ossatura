function fmtAz(a){return (a>=0?String(a).padStart(3,"0"):"-"+String(Math.abs(a)).padStart(3,"0"))+".0°"}

function grid(c,w,h,step=55){
  c.strokeStyle="rgba(120,190,240,.10)";
  for(let x=0;x<w;x+=step){c.beginPath();c.moveTo(x,0);c.lineTo(x,h);c.stroke()}
  for(let y=0;y<h;y+=40){c.beginPath();c.moveTo(0,y);c.lineTo(w,y);c.stroke()}
}
function terrain(c,w,h){
  const g=c.createLinearGradient(0,0,0,h);
  g.addColorStop(0,"rgba(0,180,255,.045)");
  g.addColorStop(.55,"rgba(15,60,70,.08)");
  g.addColorStop(1,"rgba(125,255,79,.11)");
  c.fillStyle=g;c.fillRect(0,0,w,h);
  c.strokeStyle="rgba(120,190,240,.10)";
  for(let i=0;i<8;i++){
    c.beginPath();
    for(let x=0;x<w;x+=12){
      const y=h-35-i*25+Math.sin(x*.006+i*1.7)*15+Math.sin(x*.017+i)*5;
      x?c.lineTo(x,y):c.moveTo(x,y);
    }
    c.stroke();
  }
}

}
function sector(c,cx,cy,az,spread,radius,color,alpha,label){
  const a=(az-90)*Math.PI/180,s=spread*Math.PI/180;
  c.fillStyle=color.replace("1)",alpha+")");
  c.strokeStyle=color.replace("1)",".75)");
  c.lineWidth=2;
  c.beginPath();c.moveTo(cx,cy);c.arc(cx,cy,radius,a-s/2,a+s/2);c.closePath();c.fill();c.stroke();
  c.strokeStyle=color.replace("1)",".28)");
  for(let k=1;k<9;k++){let aa=a-s/2+s*k/9;c.beginPath();c.moveTo(cx,cy);c.lineTo(cx+Math.cos(aa)*radius,cy+Math.sin(aa)*radius);c.stroke()}
  c.fillStyle=color.replace("1)","1)");
  c.font="bold 15px system-ui";
  c.fillText(label,cx+Math.cos(a)*radius*.72-45,cy+Math.sin(a)*radius*.72);
}

    armed: "site",

    evidence: []
  };

  function el(id){ return document.getElementById(id); }
  function now(){ return new Date().toLocaleTimeString(); }

    state.evidence.unshift({ts:new Date().toISOString(), action:msg, armed:state.armed});

    const x = el(id);
    if(!x) return;
    x.value = value;
    x.dispatchEvent(new Event("input",{bubbles:true}));
  }

  function setChecked(id,value){
    const x = el(id);
    if(!x) return;
    x.checked = !!value;
    x.dispatchEvent(new Event("input",{bubbles:true}));

    const r = host.getBoundingClientRect();
    const dpr = window.devicePixelRatio || 1;
    overlay.style.width = r.width + "px";
    overlay.style.height = Math.max(1,r.height-26) + "px";
    overlay.width = Math.floor(r.width * dpr);
    overlay.height = Math.floor(Math.max(1,r.height-26) * dpr);
    overlay.getContext("2d").setTransform(dpr,0,0,dpr,0,0);
  }

  function drawOverlay(id){
    const overlay = el("op_" + id);

    resizeOverlay(overlay, host);
    const ctx = overlay.getContext("2d");
    const w = overlay.clientWidth;
    const h = overlay.clientHeight;
    ctx.clearRect(0,0,w,h);

    ctx.font = "11px ui-monospace,monospace";
    ctx.fillStyle = id === state.armed ? "rgba(255,213,0,.95)" : "rgba(0,217,255,.75)";
    ctx.fillText(id.toUpperCase()+" · "+(id===state.armed?"ARMED":"READY"), 10, 18);

    marks.forEach(m=>{
      ctx.strokeStyle = m.color;
      ctx.fillStyle = m.color;
      ctx.lineWidth = 1.5;

      ctx.moveTo(m.x*w-12,m.y*h);
      ctx.lineTo(m.x*w+12,m.y*h);
      ctx.moveTo(m.x*w,m.y*h-12);
      ctx.lineTo(m.x*w,m.y*h+12);

  }

      const tilt = Math.round(((0.75-pt.y)*30)*2)/2;
      applyParam("tilt", Math.max(-10,Math.min(20,tilt)));

    Object.entries(map).forEach(([id,val])=>{if(el(id)) el(id).textContent=val});

    const rows = [
      ["VSWR","≤ 1.50",p.vswr.toFixed(2),m.qVswr],
      ["Return Loss","≥ 16 dB",p.rl.toFixed(1)+" dB",m.qRl],
      ["Isolation","≥ 28 dB",p.iso.toFixed(1)+" dB",m.qIso],

  function drawTrace(){
    const c=el("v16Trace");
    if(!c) return;
    const x=c.getContext("2d"), w=c.width, h=c.height;
    x.clearRect(0,0,w,h);
    x.strokeStyle="rgba(120,190,240,.20)";
    for(let i=0;i<w;i+=32){x.beginPath();x.moveTo(i,0);x.lineTo(i,h);x.stroke()}
    const data=v16.samples.slice().reverse();
    if(data.length<2) return;
    [["snr","#7dff4f",70],["margin","#ffd500",20]].forEach(([key,col,scale])=>{
      x.beginPath();
      data.forEach((d,i)=>{
        const px=i/(80-1)*w;
        const py=h-4-Math.max(0,Math.min(1,d[key]/scale))*(h-8);
        i?x.lineTo(px,py):x.moveTo(px,py);
      });
      x.strokeStyle=col;x.lineWidth=1.5;x.stroke();
    });
  }
