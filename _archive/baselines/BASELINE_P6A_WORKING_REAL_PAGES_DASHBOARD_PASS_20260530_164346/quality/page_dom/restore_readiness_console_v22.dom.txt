<script>
const API = '/api'

async function get(path) {
  const r = await fetch(API + path)
  if (!r.ok) throw new Error(path + ' ' + r.status)
  return await r.json()
}

function k(id, v) { document.getElementById(id).textContent = v ?? '—' }
function j(id, data) { document.getElementById(id).textContent = JSON.stringify(data, null, 2) }

function renderChecks(readiness) {
  const el = document.getElementById('checks')
  el.innerHTML = ''
  for (const c of readiness.checks || []) {
    const div = document.createElement('div')
    div.className = 'check ' + (c.passed ? 'pass' : 'fail')
    div.innerHTML = `
      <b>${c.check_id} · ${c.passed ? 'PASS' : 'FAIL'} · ${c.severity}</b>
      <span>${c.description}</span>
      <em>${JSON.stringify(c.evidence)}</em>
    `
    el.appendChild(div)
  }
}

async function loadAll() {
  const health = await get('/health')
  const readiness = await get('/restore/readiness')

  document.getElementById('badge').textContent = health.version + ' · ' + health.operational_mode

  k('k_backend', health.version)
  k('k_status', readiness.overall_status)
  k('k_checks', `${readiness.summary.passed}/${readiness.summary.total}`)
  k('k_backup', readiness.summary.latest_backup || '—')
  k('k_mode', readiness.mode)
  k('k_destructive', String(readiness.destructive_actions_enabled))

  renderChecks(readiness)
  j('output', readiness)
}

async function runDrill() {
  const drill = await get('/restore/drill')
  j('output', drill)
}

async function loadPlan() {
  const plan = await get('/restore/plan')
  j('output', plan)
}

async function verifyBackup() {
  const verify = await get('/restore/verify-backup')
  j('output', verify)
}

loadAll().catch(e => {
  document.getElementById('badge').textContent = 'ERROR'
  j('output', {error:String(e)})
})
setInterval(loadAll, 30000)
</script>
