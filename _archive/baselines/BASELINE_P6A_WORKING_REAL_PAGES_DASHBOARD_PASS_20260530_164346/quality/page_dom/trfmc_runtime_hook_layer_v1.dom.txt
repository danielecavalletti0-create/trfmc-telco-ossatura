<script>
const $ = id => document.getElementById(id);
function renderBool(v){ return v ? '<span class="oktxt">OK</span>' : '<span class="bad">MISSING</span>'; }
function item(k,v){
  const shown = typeof v === 'boolean' ? renderBool(v) : `<span>${String(v || '—')}</span>`;
  return `<div class="item"><b>${k}</b>${shown}</div>`;
}
function countOk(obj){
  return Object.values(obj || {}).filter(v => v === true || (typeof v === 'string' && v.length > 0)).length;
}
function countTotal(obj){ return Object.values(obj || {}).length; }

async function health(){
  try{
    const r = await fetch('/api/health',{cache:'no-store'});
    const j = await r.json();
    $('healthPill').textContent = j.ok ? 'Health: online' : 'Health: degraded';
  }catch(e){
    $('healthPill').textContent = 'Health: unavailable';
    $('healthPill').classList.remove('ok');
  }
}

async function loadReadiness(){
  try{
    const r = await fetch('/trfmc_runtime_readiness_latest.json',{cache:'no-store'});
    const j = await r.json();
    $('raw').textContent = JSON.stringify(j,null,2);

    $('kPortal').textContent = j.portal?.official_port || '—';
    $('kOpen5gs').textContent = `${countOk(j.open5gs)}/${countTotal(j.open5gs)}`;
    $('kUeransim').textContent = `${countOk(j.ueransim)}/${countTotal(j.ueransim)}`;
    $('kHackrf').textContent = `${countOk(j.hackrf)}/${countTotal(j.hackrf)}`;
    $('kEvidence').textContent = `${countOk(j.evidence_storage)}/${countTotal(j.evidence_storage)}`;
    $('kDisk').textContent = j.quality?.disk_free || '—';

    $('scripts').innerHTML =
      '<h4>Project scripts</h4>' + Object.entries(j.project_scripts || {}).map(([k,v])=>item(k,v)).join('') +
      '<h4>5G scripts</h4>' + Object.entries(j.fiveg_scripts || {}).map(([k,v])=>item(k,v)).join('');

    $('core').innerHTML =
      '<h4>Open5GS</h4>' + Object.entries(j.open5gs || {}).map(([k,v])=>item(k,v)).join('') +
      '<h4>UERANSIM</h4>' + Object.entries(j.ueransim || {}).map(([k,v])=>item(k,v)).join('');

    $('rf').innerHTML =
      '<h4>HackRF tooling</h4>' + Object.entries(j.hackrf || {}).map(([k,v])=>item(k,v)).join('') +
      '<h4>Evidence storage</h4>' + Object.entries(j.evidence_storage || {}).map(([k,v])=>item(k,v)).join('') +
      '<h4>Quality</h4>' + Object.entries(j.quality || {}).map(([k,v])=>item(k,v)).join('');
  }catch(e){
    $('raw').textContent = 'Readiness JSON non trovato. Esegui: ./trfmc_runtime_readiness_check_v1.sh';
  }
}

health();
loadReadiness();
