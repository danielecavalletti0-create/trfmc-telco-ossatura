<script>
"use strict";

const $ = id => document.getElementById(id);
let tick = 0;
let logLines = [];
let waterfallRows = [];

function addEvent(msg){
  const t = new Date().toLocaleTimeString();
  logLines.unshift(`[${t}] ${msg}`);
  logLines = logLines.slice(0,90);
  $("log").textContent = logLines.join("\n");
}

function n(v,d=2){ return Number(v).toFixed(d); }

function loadState(){
  try{
    const s = JSON.parse(localStorage.getItem("trfmc_fft_gapless_v1") || "{}");
    for(const k of Object.keys(s)){
      if($(k)) $(k).value = s[k];
    }
  }catch(e){}
}

function saveState(){
  const ids = ["mode","ifbw","factor","nf","fft","overlap","windowFn","water","pulse","thr"];
  const s = {};
  for(const id of ids){ s[id] = $(id).value; }
  localStorage.setItem("trfmc_fft_gapless_v1", JSON.stringify(s));
}

function resetState(){
  localStorage.removeItem("trfmc_fft_gapless_v1");
  location.reload();
}

function preset(p){
  if(p==="rs20"){
    $("mode").value="GAPLESS"; $("ifbw").value=20; $("factor").value=1.28; $("fft").value=4096; $("overlap").value=50; $("pulse").value=320; $("windowFn").value="Blackman";
  }
  if(p==="hires"){
    $("mode").value="GAPLESS"; $("ifbw").value=10; $("factor").value=1.28; $("fft").value=16384; $("overlap").value=75; $("pulse").value=900; $("windowFn").value="Hann";
  }
  if(p==="pulse"){
    $("mode").value="GAPLESS"; $("ifbw").value=40; $("factor").value=1.28; $("fft").value=2048; $("overlap").value=75; $("pulse").value=80; $("windowFn").value="Blackman";
  }
  if(p==="wide80"){
    $("mode").value="PANORAMA"; $("ifbw").value=80; $("factor").value=1.28; $("fft").value=4096; $("overlap").value=50; $("pulse").value=160; $("windowFn").value="Blackman";
  }
  addEvent("PRESET: " + p.toUpperCase() + " loaded");
  update();
}

function calc(){
  const mode = $("mode").value;
  const ifbwMHz = +$("ifbw").value;
  const factor = +$("factor").value;
  const nf = +$("nf").value;
  const fft = +$("fft").value;
  const overlap = +$("overlap").value;
  const pulseUs = +$("pulse").value;
  const thr = +$("thr").value;

  const fs = ifbwMHz * factor * 1e6;
  const binHz = fs / fft;
  const rbwKHz = binHz / 1000;
  const frameUs = fft / fs * 1e6;
  const levelPulseUs = 2 * frameUs;
  const hopUs = frameUs * (1 - overlap / 100);
  const fftPerSec = 1e6 / Math.max(hopUs, 0.001);
  const dnl = -174 + nf + 10 * Math.log10(binHz);
  const poi = pulseUs >= levelPulseUs ? "LEVEL OK" : pulseUs >= hopUs ? "DETECT OK" : "RISK";
  const reliability = pulseUs >= levelPulseUs ? 98 : pulseUs >= frameUs ? 82 : pulseUs >= hopUs ? 64 : 35;
  const bufferRisk = mode === "BUFFERED" ? Math.min(95, 35 + ifbwMHz/1.4 + fft/900) : Math.max(3, 30 - overlap/3);

  return {mode,ifbwMHz,factor,nf,fft,overlap,pulseUs,thr,fs,binHz,rbwKHz,frameUs,levelPulseUs,hopUs,fftPerSec,dnl,poi,reliability,bufferRisk};
}

function update(){
  const c = calc();

  $("vMode").textContent = c.mode;
  $("vIfbw").textContent = n(c.ifbwMHz,1);
  $("vFactor").textContent = n(c.factor,2);
  $("vNf").textContent = n(c.nf,1);
  $("vOverlap").textContent = c.overlap + "%";
  $("vWater").textContent = $("water").value + " fps";
  $("vPulse").textContent = c.pulseUs + " µs";
  $("vThr").textContent = c.thr + " dB";

  $("kMode").textContent = c.mode;
  $("kIfbw").textContent = n(c.ifbwMHz,1);
  $("kFs").textContent = n(c.fs/1e6,2);
  $("kFft").textContent = c.fft;
  $("kRbw").textContent = n(c.rbwKHz,2);
  $("kFrame").textContent = n(c.frameUs,1);
  $("kFps").textContent = Math.round(c.fftPerSec).toLocaleString();
  $("kDnl").textContent = n(c.dnl,1);
  $("kOverlap").textContent = c.overlap;
  $("kPulse").textContent = n(c.levelPulseUs,0);
  $("kPoi").textContent = c.poi;

  $("specOverlay").textContent = `IF ${n(c.ifbwMHz,1)} MHz · fs ${n(c.fs/1e6,2)} Ms/s · RBW ${n(c.rbwKHz,2)} kHz · DNL ${n(c.dnl,1)} dBm · FFT ${c.fft}`;
  $("waterOverlay").textContent = `Waterfall · hop ${n(c.hopUs,1)} µs · ${Math.round(c.fftPerSec).toLocaleString()} FFT/s · overlap ${c.overlap}%`;

  $("rbwOut").innerHTML = `fs = ${n(c.fs/1e6,3)} Ms/s<br>N = ${c.fft}<br>BWbin = ${n(c.binHz,1)} Hz = ${n(c.rbwKHz,3)} kHz`;
  $("dnlOut").innerHTML = `NF = ${n(c.nf,1)} dB<br>DNL = ${n(c.dnl,2)} dBm<br>Più piccola è la RBW, più bassa è la noise floor visualizzata.`;
  $("pulseOut").innerHTML = `Frame = ${n(c.frameUs,2)} µs<br>2×frame = ${n(c.levelPulseUs,2)} µs<br>Pulse = ${n(c.pulseUs,1)} µs<br>Status = <span class="${c.poi==='RISK'?'bad':c.poi==='DETECT OK'?'warn':'ok'}">${c.poi}</span>`;

  $("gaplessScore").textContent = c.mode === "GAPLESS" ? "OK" : c.mode === "BUFFERED" ? "N/A" : "PARTIAL";
  $("bufferRisk").textContent = c.bufferRisk > 70 ? "HIGH" : c.bufferRisk > 35 ? "MED" : "LOW";
  $("bufferRisk").className = c.bufferRisk > 70 ? "bad" : c.bufferRisk > 35 ? "warn" : "ok";
  $("levelReliability").textContent = n(c.reliability,0) + "%";
  $("levelReliability").className = c.reliability > 90 ? "ok" : c.reliability > 60 ? "warn" : "bad";

function drawSpectrum(){
  const c = $("spec"), r = c.getBoundingClientRect(), dpr = window.devicePixelRatio || 1;
  c.width = r.width*dpr; c.height = r.height*dpr;
  const x = c.getContext("2d"); x.setTransform(dpr,0,0,dpr,0,0);
  const w = r.width, h = r.height;
  const p = calc();
  x.fillStyle = "#020b12"; x.fillRect(0,0,w,h);

  x.strokeStyle = "rgba(0,210,255,.16)";
  for(let i=0;i<w;i+=44){x.beginPath();x.moveTo(i,0);x.lineTo(i,h);x.stroke();}
  for(let j=0;j<h;j+=34){x.beginPath();x.moveTo(0,j);x.lineTo(w,j);x.stroke();}

  x.strokeStyle = "#ffd400"; x.lineWidth = 1.4; x.beginPath();
  for(let i=0;i<w;i++){
    let y = h-52 + Math.sin((i+tick)/23)*4 + Math.sin((i+tick)/7)*1.2;
    const peaks = [0.16,0.28,0.44,0.58,0.73,0.86];
    for(const pp of peaks){
      const dx = i - w*pp;
      y -= Math.exp(-(dx*dx)/(2*(18+20*pp)*(18+20*pp))) * (45 + 38*Math.sin(tick/50+pp*6));
    }
    if(p.pulseUs < p.levelPulseUs && Math.abs(i - (w*((tick%220)/220))) < 5){
      y -= 90;
    }
    if(i===0)x.moveTo(i,y); else x.lineTo(i,y);
  }
  x.stroke();

  x.fillStyle = "rgba(0,229,255,.10)";
  const marker = (tick % 260) / 260 * w;
  x.fillRect(marker,0,10,h);
  x.strokeStyle = "rgba(255,61,127,.8)";
  x.strokeRect(marker,0,10,h);

  x.fillStyle="#7fa3b7"; x.font="10px monospace";
  for(let k=0;k<6;k++) x.fillText((-20-k*20)+" dBm",w-62,20+k*42);
}

function drawWaterfall(){
  const c = $("waterfall"), r = c.getBoundingClientRect(), dpr = window.devicePixelRatio || 1;
  c.width = r.width*dpr; c.height = r.height*dpr;
  const x = c.getContext("2d"); x.setTransform(dpr,0,0,dpr,0,0);
  const w = Math.floor(r.width), h = Math.floor(r.height);

  const row = [];
  for(let i=0;i<w;i++){
    let v = 12 + Math.sin((i+tick)/19)*7 + Math.random()*8;
    for(const pp of [.18,.37,.53,.69,.84]){
      const dx=i-w*pp;
      v += Math.exp(-(dx*dx)/(2*90))*120;
    }
    if(Math.abs(i - (w*((tick%200)/200))) < 4) v += 180;
    row.push(Math.min(255,Math.max(0,v)));
  }
  waterfallRows.unshift(row);
  waterfallRows = waterfallRows.slice(0,h);

  const img = x.createImageData(w,h);
  for(let y=0;y<h;y++){
    const src = waterfallRows[y] || row;
    for(let i=0;i<w;i++){
      const v = src[i] || 0;
      const idx=(y*w+i)*4;
      img.data[idx] = Math.max(0, v-80);
      img.data[idx+1] = Math.min(255, v*1.1);
      img.data[idx+2] = Math.min(255, 80+v*1.4);
      img.data[idx+3] = 255;
    }
  }
  x.putImageData(img,0,0);
}

}
