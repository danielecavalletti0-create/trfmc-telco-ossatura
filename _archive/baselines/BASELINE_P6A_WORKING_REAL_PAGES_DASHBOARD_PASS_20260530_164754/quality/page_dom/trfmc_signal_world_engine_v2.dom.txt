<script>
const $ = id => document.getElementById(id);

const bands = [
  {v:"27", label:"27 MHz", cls:"HF/CB", mod:"AM", span:15},
  {v:"144", label:"144 MHz", cls:"VHF voice/data", mod:"FM", span:28},
  {v:"244", label:"244 MHz", cls:"VHF/UHF utility", mod:"FM", span:34},
  {v:"433", label:"433 MHz", cls:"ISM remote / FSK", mod:"FSK Remote", span:26},
  {v:"868", label:"868 MHz", cls:"ISM / IoT", mod:"FSK Remote", span:38},
  {v:"1800", label:"1.8 GHz", cls:"LTE-like OFDM", mod:"OFDM", span:55},
  {v:"2400", label:"2.4 GHz", cls:"WiFi / ISM", mod:"WiFi-like", span:74},
  {v:"3500", label:"3.5 GHz", cls:"5G NR / OFDM", mod:"OFDM", span:80},
  {v:"5800", label:"5.8 GHz", cls:"WiFi / ISM", mod:"WiFi-like", span:86}
];

const world = $("worldCanvas");
const spec = $("spectrumCanvas");
const waterfall = $("waterfallCanvas");
const iq = $("iqCanvas");
const cons = $("constellationCanvas");

const wctx = world.getContext("2d");
const sctx = spec.getContext("2d");
const wfctx = waterfall.getContext("2d");
const iqctx = iq.getContext("2d");
const cctx = cons.getContext("2d");

let t = 0;
let emitters = [];
let lastBand = "3500";
let lastLog = 0;

function initBandButtons(){
  const box = $("bandButtons");
  bands.forEach(b=>{
    const btn = document.createElement("button");
    btn.textContent = b.label;
    btn.dataset.v = b.v;
    btn.onclick = () => {
      $("bandSelect").value = b.v;
      $("span").value = b.span;
      $("modulation").value = b.mod;
      regenerateEmitters();
    };
    box.appendChild(btn);
  });
}

function currentBand(){
  return bands.find(b=>b.v === $("bandSelect").value) || bands[7];
}

function params(){
  const b = currentBand();
  return {
    band: b,
    freq: Number(b.v),
    span: 5 + Number($("span").value)/100*115,
    gain: 2 + Number($("gain").value)/100*48,
    noise: -110 + Number($("noise").value)/100*50,
    density: Number($("density").value),
    mode: $("scanMode").value,
    mod: $("modulation").value
  };
}

function updateLabels(){
  const p = params();
  $("bandLabel").textContent = p.band.label;
  $("scanLabel").textContent = $("scanMode").selectedOptions[0].textContent;
  $("spanLabel").textContent = Math.round(p.span) + " MHz";
  $("gainLabel").textContent = Math.round(p.gain) + " dB";
  $("noiseLabel").textContent = Math.round(p.noise) + " dBm";
  $("densityLabel").textContent = Math.round(p.density) + "%";
  $("modLabel").textContent = p.mod;

  $("kBand").textContent = p.band.label;
  $("kMode").textContent = p.mode.toUpperCase();
  $("kClass").textContent = classifyText(p);
  document.querySelectorAll("#bandButtons button").forEach(btn=>btn.classList.toggle("active", btn.dataset.v === String(p.freq)));
}

function classifyText(p){
  if(p.mod === "OFDM") return p.freq >= 3000 ? "5G NR / OFDM" : "LTE-like OFDM";
  if(p.mod === "WiFi-like") return "WiFi / OFDM burst";
  if(p.mod === "FSK Remote") return "FSK / remote burst";
  if(p.freq <= 30) return "HF / AM voice";
  if(p.mod === "FM") return "FM voice/data";
  return p.mod + " signal";
}

function regenerateEmitters(){
  const p = params();
  emitters = [];
  const count = Math.max(5, Math.round(6 + p.density/7));
  for(let i=0;i<count;i++){
    emitters.push({
      x: 120 + Math.random()*(world.width-240),
      y: 120 + Math.random()*(world.height-230),
      power: 0.25 + Math.random()*0.75,
      phase: Math.random()*Math.PI*2,
      drift: (Math.random()-.5)*0.35,
      type: Math.random() < 0.13 ? "jammer" : "signal",
      freqOffset: (Math.random()-.5)*0.8
    });
  }
  lastBand = $("bandSelect").value;
}

function grid(ctx,w,h){
  ctx.strokeStyle = "rgba(120,190,255,.075)";
  ctx.lineWidth = 1;
  for(let x=0;x<w;x+=50){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,h);ctx.stroke();}
  for(let y=0;y<h;y+=40){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(w,y);ctx.stroke();}
}

function label(ctx,txt,x,y,color="rgba(234,243,255,.82)",size=14){
  ctx.fillStyle = color;
  ctx.font = `${size}px system-ui`;
  ctx.fillText(txt,x,y);
}

function drawWorld(){
  const p = params();
  const ctx = wctx;
  const w = world.width, h = world.height;
  ctx.clearRect(0,0,w,h);

function generateSpectrum(){
  const p = params();
  const w = spec.width;
  const arr = new Float32Array(w);
  let peak = -140;

  for(let x=0;x<w;x++){
    let db = p.noise + (Math.random()-.5)*4 + Math.sin(x*.02+t*.025)*1.8;

    emitters.forEach((e,i)=>{
      const center = w*(0.5 + e.freqOffset*0.42 + Math.sin(t*.003+i)*0.018);
      let width = 10 + p.span*.18 + e.power*26;
      let amp = (12+p.gain*.45)*e.power*(p.density/100);
      if(e.type === "jammer"){ width *= 3.2; amp *= 1.28; }
      db += Math.exp(-Math.pow(x-center,2)/(width*width))*amp;
    });

    if(p.mode === "zerospan"){
      const gate = Math.sin(t*.08) > 0 ? 1 : .25;
      db += Math.exp(-Math.pow(x-w*.5,2)/(40*40))*gate*22;
    }

    if(p.mod === "OFDM" || p.mod === "WiFi-like"){
      for(let k=-12;k<=12;k++){
        const c = w*.5 + k*(5+p.span*.055);
        db += Math.exp(-Math.pow(x-c,2)/(9+p.span*.04))*5.4;
      }
    }

    if(p.mod === "FSK Remote"){
      db += Math.exp(-Math.pow(x-w*.46,2)/(14*14))*12;
      db += Math.exp(-Math.pow(x-w*.54,2)/(14*14))*12;
    }

    if(p.mod === "AM"){
      db += Math.exp(-Math.pow(x-w*.5,2)/(10*10))*14;
      db += Math.exp(-Math.pow(x-w*.43,2)/(15*15))*8;
      db += Math.exp(-Math.pow(x-w*.57,2)/(15*15))*8;
    }

    arr[x] = db;
    peak = Math.max(peak,db);
  }

  return {arr,peak};
}

function constellationPoints(){
  const p=params();
  const pts=[];
  const n=420;
  const noise=Math.max(.015,(100-Number($("gain").value))/550 + Number($("noise").value)/2800);

  const add=(x,y)=>pts.push([x+(Math.random()-.5)*noise*2,y+(Math.random()-.5)*noise*2]);

  for(let i=0;i<n;i++){
    if(p.mod==="BPSK") add(Math.random()>.5?.75:-.75,0);
    else if(p.mod==="QPSK") add(Math.random()>.5?.62:-.62,Math.random()>.5?.62:-.62);
    else if(p.mod==="16QAM"){
      const v=[-.75,-.25,.25,.75]; add(v[Math.floor(Math.random()*4)],v[Math.floor(Math.random()*4)]);
    }else if(p.mod==="OFDM" || p.mod==="WiFi-like"){
      const a=Math.random()*Math.PI*2, r=.18+Math.random()*.72; add(Math.cos(a)*r,Math.sin(a)*r);
    }else if(p.mod==="FSK Remote"){
      add(Math.random()>.5?.55:-.55, Math.sin(i*.4+t*.04)*.18);
    }else{
      const a=i*.19+t*.025, r=p.mod==="AM"?.35+Math.sin(i*.08+t*.02)*.22:.62; add(Math.cos(a)*r,Math.sin(a)*r);
    }
  }
  return pts;
}
