<script>
let registry=null;
let currentFilter='all';

const priorityNames = [
  'trfmc_official_safe_entrypoint_v6r3_command_center.html',
  'rfpro_unified_console.html',
  'trfmc_signal_intelligence_center_v1.html',
  'trfmc_3d_rf_asset_renderer_webgl_v2r2_reality.html',
  'trfmc_core_network_live_ops_bridge_v1.html',
  'trfmc_rf_metrology_calibration_lab_v1.html',
  'trfmc_antenna_system_explorer_STABLE_CLEAN_RECOVERY.html',
  'trfmc_rf_microwave_engineering_v1.html',
  'trfmc_portal_link_graph_v1.html'
];

function esc(s){
  return String(s ?? '').replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}

function badgeFor(x){
  let arr=[];
  arr.push(`<span class="badge">${esc(x.class)}</span>`);
  if(x.webgl) arr.push('<span class="badge ok">WEBGL</span>');
  if(x.core_api) arr.push('<span class="badge warn">API</span>');
  if(x.has_iframe) arr.push('<span class="badge warn">IFRAME</span>');
  if(x.external_refs) arr.push('<span class="badge bad">EXT</span>');
  if(priorityNames.includes(x.name)) arr.push('<span class="badge violet">PRIORITY</span>');
  return arr.join('');
}

function moduleCard(x, extra=''){
  const official = x.class === 'official_shell' ? ' official' : '';
  const priority = priorityNames.includes(x.name) ? ' priority' : '';
  return `
  <article class="module${official}${priority} ${extra}" data-class="${esc(x.class)}" data-name="${esc(x.name).toLowerCase()}">
    <h3 title="${esc(x.name)}">${esc(x.name)}</h3>
    <div>${badgeFor(x)}</div>
    <div class="meta">
      size: ${esc(x.size)} bytes<br>
      refs: ${esc(x.refs_count)} · url: ${esc(x.url)}
    </div>
    <div class="moduleActions">
      <a href="${esc(x.url)}" target="_blank">Apri</a>
      <a href="/trfmc_portal_registry_unified.json" target="_blank">Registry</a>
    </div>
  </article>`;
}

function renderCounts(){
  const c=registry.counts || {};
  const labels=[
    ['total_html','Totale HTML'],
    ['official_shell','Official shell'],
    ['service','Service'],
    ['leaf_operational_candidate','Operativi'],
    ['shell_or_legacy_container','Shell legacy'],
    ['orphan_or_legacy_candidate','Orfani/legacy']
  ];
  document.getElementById('counts').innerHTML=labels.map(([k,label])=>`
    <div class="count">
      <small>${label}</small>
      <b>${c[k] ?? 0}</b>
    </div>
  `).join('');
}

function applyFilters(){
  const q=document.getElementById('search').value.trim().toLowerCase();
  const cards=[...document.querySelectorAll('.module')];
  let visible=0;
  cards.forEach(el=>{
    const cls=el.dataset.class;
    const name=el.dataset.name;
    const okFilter=currentFilter==='all' || cls===currentFilter;
    const okSearch=!q || name.includes(q) || cls.toLowerCase().includes(q);
    const show=okFilter && okSearch;
    el.classList.toggle('hidden',!show);
    if(show) visible++;
  });
  document.getElementById('visibleCount').textContent=`visibili: ${visible}`;
}

function render(){
  renderCounts();
  const pages=registry.pages || [];
  const priority=priorityNames
    .map(n=>pages.find(p=>p.name===n))
    .filter(Boolean);

  document.getElementById('priority').innerHTML = priority.length
    ? priority.map(x=>moduleCard(x)).join('')
    : '<div class="card">Nessun modulo prioritario trovato nel registry.</div>';

  document.getElementById('pages').innerHTML = pages.map(x=>moduleCard(x)).join('');
  applyFilters();
}

async function boot(){
  const r=await fetch('/trfmc_portal_registry_unified.json',{cache:'no-store'});
  registry=await r.json();
  render();

  document.getElementById('search').addEventListener('input',applyFilters);
  document.querySelectorAll('[data-filter]').forEach(btn=>{
    btn.addEventListener('click',()=>{
      currentFilter=btn.dataset.filter;
      applyFilters();
    });
  });
}

boot().catch(e=>{
  document.body.innerHTML += '<pre style="color:#ff4f7b;padding:20px">'+esc(e.stack || e)+'</pre>';
});
