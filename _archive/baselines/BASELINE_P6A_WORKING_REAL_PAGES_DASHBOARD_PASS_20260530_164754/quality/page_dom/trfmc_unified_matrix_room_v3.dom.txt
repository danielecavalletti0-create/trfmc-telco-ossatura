  const box=$("routes"); box.innerHTML="";
  Object.entries(R).forEach(([k,r])=>{
    const h=health[k]; const cls=h===200?"ok":h?"bad":"unk"; const label=h===200?"HTTP 200":h?"HTTP "+h:"not checked";

  });

  $("dom").textContent=r.d; $("sub").textContent=r.s;

  $("secondary").textContent=R[slots[1]].d;

  $("cap"+i).textContent=r.d;

function execCmd(){
  const c=$("cmd").value.trim().toUpperCase();
  if(c.includes("FULL")) presetMatrix("full");
  else if(c.includes("RF")) presetMatrix("rf");
  else if(c.includes("CORE")) presetMatrix("core");
  else if(c.includes("QUAD")) setMatrix("quad");
  else if(c.includes("DUAL")) setMatrix("dual");
  else if(c.includes("TRIPLE")) setMatrix("triple");
  else if(c.includes("SINGLE")) setMatrix("single");
  else if(c.includes("FOCUS")) layout("focus");
  else if(c.includes("WALL")) layout("wall");
  else if(c.includes("KIOSK")) layout("kiosk");
  else if(c.includes("NORMAL")) layout("normal");
  else if(c.includes("HEALTH")) preflight();
  else if(c.includes("RESET")) resetBrokenStorage();
  else if(c.includes("PING")) postPing();
  else log("unknown command → "+c);
}
