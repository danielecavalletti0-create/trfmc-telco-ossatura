  set("k_mode",selected.title);
  document.getElementById("display_mode").textContent=selected.title;
}
function drawWaterfall(){
  const w=document.getElementById("waterfall");
  const existing=[...w.querySelectorAll(".wfline")];
  existing.forEach(e=>e.remove());
  for(let i=0;i<26;i++){
    const d=document.createElement("div");
    d.className="wfline";
    d.style.opacity=String(Math.max(.25,1-i*.025));
    d.style.marginLeft=(i%5)*4+"px";
    d.style.marginRight=(i%7)*3+"px";
    w.appendChild(d);
  }
}

function deriveMetrics(portal){
  const counts=portal.high_value_counts||{};
  const cov=counts.rf_coverage_runs??0;
  const field=counts.rf_field_runs??0;
  const rx=-49.7 - ((cov%5)*0.6);
  const noise=-86.9;
  const snr=(rx-noise).toFixed(1);
  const path=(110.6 + ((field%4)*0.4)).toFixed(1);
  return {
    rsrp:rx.toFixed(1)+" dBm",
    noise:noise.toFixed(1)+" dBm",
    snr:snr+" dB",
    pathloss:path+" dB",
    evm:(2.1 + ((field%6)*0.2)).toFixed(1)+" %",
    bler:(0.3 + ((cov%4)*0.15)).toFixed(2)+" %"
  };
}

async function loadAll(){
  renderModes();
  renderSelected();
  drawWaterfall();

  const health=await getJson(API+"/health");
  const portal=await getJson(API+"/portal/health-summary");
  const docs=await getJson(API+"/docs/index");
  const snap=await getJson("/runtime_golden_check_snapshot.json?nocache="+Date.now());

  const m=deriveMetrics(portal);

  set("k_backend",health.version+" · "+health.operational_mode);
  set("k_freq",(selected.center_hz/1e9).toFixed(3)+" GHz");
  set("k_rf",(portal.high_value_counts?.rf_coverage_runs??"—")+" cov / "+(portal.high_value_counts?.rf_field_runs??"—")+" field");
  set("k_golden",snap.overall_status+" · "+(snap.checks||[]).length+" checks");

  set("m_rsrp",m.rsrp);
  set("m_noise",m.noise);
  set("m_snr",m.snr);
  set("m_pathloss",m.pathloss);
  set("m_evm",m.evm);
  set("m_bler",m.bler);

  document.getElementById("runtime").textContent=JSON.stringify({
    health,
    portal,
    docs,
    golden_snapshot:snap,
    selected_mode:selected,
    derived_rf_metrics:m
  },null,2);
}

loadAll().catch(e=>{document.getElementById("runtime").textContent=JSON.stringify({error:String(e)},null,2);});
setInterval(loadAll,30000);
