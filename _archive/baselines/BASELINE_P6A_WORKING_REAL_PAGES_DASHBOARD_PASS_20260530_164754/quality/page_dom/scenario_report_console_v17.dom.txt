<script>
const API = '/api'
let currentReport = null

async function get(path) {
  const r = await fetch(API + path)
  if (!r.ok) throw new Error(path + ' ' + r.status)
  return await r.json()
}

function k(id, v) { document.getElementById(id).textContent = v ?? '—' }
function j(id, data) { document.getElementById(id).textContent = JSON.stringify(data, null, 2) }

function render(report) {
  currentReport = report
  const s = report.executive_summary || {}
  document.getElementById('runInput').value = report.run_id || ''

  k('k_scenario', report.scenario_id)
  k('k_run', report.run_id)
  k('k_corr', report.correlation_id)
  k('k_cov', s.coverage_classification)
  k('k_timeline', s.related_timeline_entries)

  const el = document.getElementById('summary')
  el.innerHTML = `
    <div class="card"><b>Scenario</b><span>${report.scenario_id || '—'}</span><em>${report.title || ''}</em></div>
    <div class="card"><b>Run / Correlation</b><span>${report.run_id || '—'}</span><em>${report.correlation_id || '—'}</em></div>
    <div class="card"><b>RF Field</b><span>${s.rf_field_region || '—'}</span><em>Gain: ${s.antenna_gain_to_target_dbi ?? '—'} dBi</em></div>
    <div class="card"><b>Coverage</b><span>${s.coverage_classification || '—'} · ${s.los_state || '—'}</span><em>SNR: ${s.snr_db ?? '—'} dB</em></div>
    <div class="card"><b>Evidence</b><span>CloudEvents: ${s.related_cloud_events ?? '—'} · Timeline: ${s.related_timeline_entries ?? '—'}</span><em>Graph nodes: ${s.mission_graph_nodes ?? '—'} · edges: ${s.mission_graph_edges ?? '—'}</em></div>
    <div class="card"><b>Assessment</b><span>${s.assessment || '—'}</span></div>
  `
  j('report', report)
}

async function loadLatest() {
  const health = await get('/health')
  k('k_backend', health.version)
  document.getElementById('badge').textContent = health.version + ' · ' + health.operational_mode

  const report = await get('/reports/latest')
  render(report)
}

async function loadByRun() {
  const run = document.getElementById('runInput').value.trim()
  if (!run) return loadLatest()
  const report = await get('/reports/scenario/' + encodeURIComponent(run))
  render(report)
}

function exportHtml() {
  const run = (currentReport && currentReport.run_id) || document.getElementById('runInput').value.trim()
  if (!run) {
    alert('Run ID non disponibile')
    return
  }
  window.open(API + '/reports/export/html/' + encodeURIComponent(run), '_blank')
}

loadLatest().catch(e => {
  document.getElementById('badge').textContent = 'ERROR'
  j('report', {error:String(e)})
})
setInterval(loadLatest, 30000)
</script>
