function regenerateEmitters(){
  const p = params();
  const base = p.band >= 2400 ? 22 : p.band >= 433 ? 16 : 11;
  emitters = [];
  for(let i=0;i<base;i++){
    emitters.push({
      x:(Math.random()-.5)*2.2,
      y:(Math.random()-.5)*1.35,
      z:Math.random()*2.4-.35,
      a:.20+Math.random()*.90,

  lastBand = String(p.band);
}

function project(p,w,h){
  const z = p.z + 3.0;
  const scale = 450 / z;
  return {
    x:w/2 + p.x*scale,
    y:h/2 + p.y*scale - p.z*58,
    s:scale/270,
    z
  };
}

function sampleWave(i,phase=0){
  const p = params();
  const x = i*.034 + t*.055 + phase;
  let y = 0;
  if(p.wave==="Sine") y=Math.sin(x);
  else if(p.wave==="Square") y=Math.sin(x)>0?1:-1;
  else if(p.wave==="Saw") y=((x/Math.PI)%2)-1;
  else if(p.wave==="Pulse") y=Math.sin(x)>0.72?1:0;
  else if(p.wave==="Chirp") y=Math.sin(x + Math.sin(t*.01)*i*.006);
  else if(p.wave==="Burst") y=Math.sin(x)*(Math.sin(t*.025)>0?.95:.18);
  else y=(Math.random()-.5)*2;
  return y*p.amp/100;
}

function spectrumData(){
  const p=params(), w=C.spectrum.width;
  const arr=new Float32Array(w);
  let peak=-140;
  for(let i=0;i<w;i++){
    let db=p.noise+(Math.random()-.5)*4+Math.sin(i*.018+t*.022)*2;
    emitters.forEach((e,k)=>{
      const center=w*(.5+e.off*.42+Math.sin(t*.004+k)*.015);
      let width=11+p.span*.16+e.a*24;
      let amp=(14+p.gain*.45)*e.a;
      if(e.type==="anomaly"){width*=3.1;amp*=1.35;}
      db+=Math.exp(-Math.pow(i-center,2)/(width*width))*amp;
    });
    if(["OFDM","WiFi-like","5GNR-like"].includes(p.mod)){
      for(let k=-15;k<=15;k++){
        const c=w*.5+k*(5+p.span*.044);
        db+=Math.exp(-Math.pow(i-c,2)/(8+p.span*.04))*5.2;
      }
    }
    if(p.mod==="FSK"){
      db+=Math.exp(-Math.pow(i-w*.45,2)/(13*13))*13;
      db+=Math.exp(-Math.pow(i-w*.55,2)/(13*13))*13;
    }
    if(p.mod==="AM"){
      db+=Math.exp(-Math.pow(i-w*.5,2)/(10*10))*14;
      db+=Math.exp(-Math.pow(i-w*.42,2)/(16*16))*8;
      db+=Math.exp(-Math.pow(i-w*.58,2)/(16*16))*8;
    }
    arr[i]=db; peak=Math.max(peak,db);
  }
  return {arr,peak};
}

function constellationPts(){
  const p=params(), pts=[];
  const n=430;
  const noise=Math.max(.015,(100-Number($("gain").value))/620+Number($("noise").value)/3200);
  const add=(a,b)=>pts.push([a+(Math.random()-.5)*noise*2,b+(Math.random()-.5)*noise*2]);
  for(let i=0;i<n;i++){
    if(p.mod==="BPSK") add(Math.random()>.5?.75:-.75,0);
    else if(p.mod==="QPSK") add(Math.random()>.5?.62:-.62,Math.random()>.5?.62:-.62);
    else if(p.mod==="16QAM"){
      const v=[-.75,-.25,.25,.75]; add(v[Math.floor(Math.random()*4)],v[Math.floor(Math.random()*4)]);
    }else if(["OFDM","WiFi-like","5GNR-like"].includes(p.mod)){
      const a=Math.random()*Math.PI*2,r=.15+Math.random()*.78; add(Math.cos(a)*r,Math.sin(a)*r);
    }else if(p.mod==="FSK"){
      add(Math.random()>.5?.55:-.55,Math.sin(i*.4+t*.04)*.18);
    }else{
      const a=i*.19+t*.025,r=p.mod==="AM"?.35+Math.sin(i*.08+t*.02)*.22:.62; add(Math.cos(a)*r,Math.sin(a)*r);
    }
  }
  return pts;
}
