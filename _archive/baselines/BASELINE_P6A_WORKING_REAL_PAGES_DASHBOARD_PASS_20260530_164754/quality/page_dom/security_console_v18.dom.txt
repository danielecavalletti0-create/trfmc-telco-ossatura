<script>
const API = '/api'

async function get(path) {
  const r = await fetch(API + path)
  if (!r.ok) throw new Error(path + ' ' + r.status)
  return await r.json()
}

function k(id, v) { document.getElementById(id).textContent = v ?? '—' }
function j(id, data) { document.getElementById(id).textContent = JSON.stringify(data, null, 2) }

function renderChecks(posture) {
  const el = document.getElementById('checks')
  el.innerHTML = ''
  for (const c of posture.checks || []) {
    const div = document.createElement('div')
    div.className = 'check ' + (c.passed ? 'pass' : 'fail')
    div.innerHTML = `
      <b>${c.check_id} · ${c.passed ? 'PASS' : 'FAIL'} · ${c.severity}</b>
      <span>${c.description}</span>
      <em>${JSON.stringify(c.evidence)}</em>
    `
    el.appendChild(div)
  }
}

async function loadAll() {
  const health = await get('/health')
  const posture = await get('/security/posture')
  const policy = await get('/security/access-policy')
  const readiness = await get('/security/readiness')
  const runtime = await get('/security/runtime-checks')
  const audit = await get('/security/audit-log?limit=30')

  document.getElementById('badge').textContent = health.version + ' · ' + health.operational_mode

  k('k_backend', health.version)
  k('k_posture', posture.overall_status)
  k('k_mode', posture.operational_mode)
  k('k_restricted', String(posture.restricted_enabled))
  k('k_checks', `${posture.summary.passed}/${posture.summary.total}`)
  k('k_ready', readiness.decision)

  renderChecks(posture)
  j('policy', policy)
  j('readiness', readiness)
  j('runtime', {runtime, audit})
}

loadAll().catch(e => {
  document.getElementById('badge').textContent = 'ERROR'
  j('runtime', {error:String(e)})
})
setInterval(loadAll, 30000)
</script>
