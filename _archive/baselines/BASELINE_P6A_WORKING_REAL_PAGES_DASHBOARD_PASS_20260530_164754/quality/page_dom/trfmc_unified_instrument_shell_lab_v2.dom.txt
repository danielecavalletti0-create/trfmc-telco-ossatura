function log(m){
  const d=document.createElement("div");
  d.textContent="["+new Date().toLocaleTimeString()+"] "+m;
  $("log").prepend(d);
  while($("log").children.length>70) $("log").removeChild($("log").lastChild);

function renderRoutes(){
  const box=$("routes");
  box.innerHTML="";
  Object.entries(routes).forEach(([k,r])=>{
    const h = health[k];
    const cls = h===200 ? "ok" : h ? "bad" : "unk";
    const label = h===200 ? "HTTP 200" : h ? "HTTP "+h : "not checked";
    const div=document.createElement("div");
    div.className="route"+(k===active?" active":"");
    div.onclick=()=>loadRoute(k);
    div.innerHTML="<b>"+r.domain+"</b><span>"+r.title+"</span><span>"+r.url+"</span><small class='"+cls+"'>"+label+"</small>";
    box.appendChild(div);
  });
}

  $("rbw").textContent=r.rbw;

function loadRoute(k){
  if(!routes[k]) return log("blocked non-whitelisted route → "+k);
  active=k;
  const r=routes[k];
  $("frame1").src=r.url;
  $("stageTitle").textContent=r.title;
  $("stageUrl").textContent=r.url;

}

async function preflight(){
  log("preflight started");
  for(const [k,r] of Object.entries(routes)){
    try{
      const res=await fetch(r.url,{method:"HEAD",cache:"no-store"});
      health[k]=res.status;
      log("health "+k+" → HTTP "+res.status);
    }catch(e){
      health[k]="ERR";
      log("health "+k+" → ERR");
    }
  }
  try{
    const res=await fetch("/api/health",{cache:"no-store"});
    log("/api/health → HTTP "+res.status);
  }catch(e){
    log("/api/health → ERR");
  }
  renderRoutes();
}

function execCmd(){
  const c=$("cmd").value.trim().toUpperCase();
  if(c.includes("ANTENNA")) loadRoute("antenna");
  else if(c.includes("DSP")) loadRoute("dsp");
  else if(c.includes("WIFI") || c.includes("QAM")) loadRoute("wifi");
  else if(c.includes("CORE") || c.includes("AKA")) loadRoute("core");
  else if(c.includes("NOC")) loadRoute("noc");
  else if(c.includes("WAR")) loadRoute("war");
  else if(c.includes("MASTER")) loadRoute("master");
  else if(c.includes("FOCUS")) setLayout("focus");
  else if(c.includes("WALL")) setLayout("wall");
  else if(c.includes("KIOSK")) setLayout("kiosk");
  else if(c.includes("NORMAL")) setLayout("normal");
  else if(c.includes("SPLIT")) toggleSplit();
  else if(c.includes("HEALTH")) preflight();
  else if(c.includes("RESET")) resetBrokenStorage();
  else if(c.includes("PING")) postPing();
  else log("unknown command → "+c);
}
