<script>
"use strict";
const $ = id => document.getElementById(id);
let tick = 0;
let logLines = [];

function n(v,d=3){return Number(v).toFixed(d)}
function addEvent(msg){
  const t = new Date().toLocaleTimeString();
  logLines.unshift(`[${t}] ${msg}`);
  logLines = logLines.slice(0,90);
  $("log").textContent = logLines.join("\n");
}

function loadState(){
  try{
    const s = JSON.parse(localStorage.getItem("trfmc_rf_metrology_v1") || "{}");
    for(const k of Object.keys(s)){ if($(k)) $(k).value = s[k]; }
  }catch(e){}
}
function saveState(){
  const ids = ["mode","freq","gen","att","cable","cal","vswr","drift","rep","ref","res"];
  const s = {};
  for(const id of ids){ s[id] = $(id).value; }
  localStorage.setItem("trfmc_rf_metrology_v1", JSON.stringify(s));
}
function resetState(){ localStorage.removeItem("trfmc_rf_metrology_v1"); location.reload(); }

function preset(p){
  if(p==="trfl"){ $("mode").value="TRFL"; $("freq").value=2.4; $("gen").value=0; $("att").value=30; $("cable").value=1.2; $("cal").value=0.08; $("vswr").value=0.02; }
  if(p==="attenuator"){ $("mode").value="ATTENUATOR"; $("freq").value=10; $("gen").value=10; $("att").value=60; $("cable").value=2.1; $("cal").value=0.04; $("vswr").value=0.06; }
  if(p==="phase"){ $("mode").value="PHASE_NOISE"; $("freq").value=1; $("gen").value=13; $("att").value=10; $("drift").value=0.01; }
  if(p==="mod"){ $("mode").value="AM_FM_PM"; $("freq").value=0.1; $("gen").value=-10; $("att").value=3; $("rep").value=0.025; }
  addEvent("PRESET: " + p.toUpperCase() + " loaded");
  update();
}

function calc(){
  const mode = $("mode").value;
  const freq = +$("freq").value;
  const gen = +$("gen").value;
  const att = +$("att").value;
  const cable = +$("cable").value;
  const cal = +$("cal").value;
  const vswr = +$("vswr").value;
  const drift = +$("drift").value;
  const rep = +$("rep").value;
  const ref = +$("ref").value;
  const res = +$("res").value;

  const raw = gen - att - cable;
  const corr = raw + cal + cable + vswr;
  const attenMeasured = gen - corr;
  const attResidual = attenMeasured - att;

  const step10 = att / 10;
  const linTol = 0.009 + 0.005 * step10;
  const linPass = Math.abs(attResidual) <= Math.max(0.04, linTol + 0.02);

  const pn = -105 - 8*Math.log10(Math.max(freq,0.01)) - 4*Math.log10(Math.max(att,1)) - (mode==="PHASE_NOISE"?8:0);
  const xcorrImprove = Math.min(20, 5 + Math.log2(Math.max(att,1))*2.2);
  const pnXcorr = pn - xcorrImprove;

  const uRep = rep;
  const uRef = ref;
  const uRes = res / Math.sqrt(3);
  const uDrift = drift / Math.sqrt(3);
  const uVswr = vswr / Math.sqrt(3);
  const uc = Math.sqrt(uRep*uRep + uRef*uRef + uRes*uRes + uDrift*uDrift + uVswr*uVswr);
  const U = 2 * uc;

  const am = Math.abs(0.12 + Math.sin(freq)*0.08 + drift*0.5);
  const fm = Math.abs(0.18 + Math.cos(freq)*0.10 + vswr*0.7);
  const pm = Math.abs(0.10 + Math.sin(att/10)*0.07 + rep*2);

  return {mode,freq,gen,att,cable,cal,vswr,drift,rep,ref,res,raw,corr,attenMeasured,attResidual,linTol,linPass,pn,pnXcorr,xcorrImprove,uRep,uRef,uRes,uDrift,uVswr,uc,U,am,fm,pm};
}

function update(){
  const c = calc();

  $("vMode").textContent = c.mode;
  $("vFreq").textContent = n(c.freq,3);
  $("vGen").textContent = n(c.gen,1);
  $("vAtt").textContent = n(c.att,1);
  $("vCable").textContent = n(c.cable,2);
  $("vCal").textContent = n(c.cal,2);
  $("vVswr").textContent = n(c.vswr,3);
  $("vDrift").textContent = n(c.drift,3);
  $("vRep").textContent = n(c.rep,3);
  $("vRef").textContent = n(c.ref,3);
  $("vRes").textContent = n(c.res,3);

  $("kMode").textContent = c.mode;

function drawLevel(){
  const canvas = $("levelCanvas"), r = canvas.getBoundingClientRect(), dpr = devicePixelRatio || 1;
  canvas.width = r.width*dpr; canvas.height = r.height*dpr;
  const x = canvas.getContext("2d"); x.setTransform(dpr,0,0,dpr,0,0);
  const w = r.width, h = r.height, c = calc();

  x.fillStyle="#020b12"; x.fillRect(0,0,w,h);
  x.strokeStyle="rgba(0,210,255,.16)";
  for(let i=0;i<w;i+=44){x.beginPath();x.moveTo(i,0);x.lineTo(i,h);x.stroke();}
  for(let j=0;j<h;j+=34){x.beginPath();x.moveTo(0,j);x.lineTo(w,j);x.stroke();}

  const levels = [
    ["GEN", c.gen],
    ["ATT OUT RAW", c.raw],
    ["CAL FACTOR", c.raw + c.cal],
    ["CABLE CORR", c.raw + c.cal + c.cable],
    ["VSWR CORR", c.corr]
  ];
  const min = Math.min(...levels.map(v=>v[1])) - 5;
  const max = Math.max(...levels.map(v=>v[1])) + 5;
  const mapY = v => h - 42 - (v-min)/(max-min)*(h-80);

  x.strokeStyle="#ffd400"; x.lineWidth=2; x.beginPath();
  levels.forEach((p,i)=>{
    const xx=60+i*(w-120)/(levels.length-1), yy=mapY(p[1]);
    if(i===0)x.moveTo(xx,yy); else x.lineTo(xx,yy);
  });
  x.stroke();

  levels.forEach((p,i)=>{
    const xx=60+i*(w-120)/(levels.length-1), yy=mapY(p[1]);
    x.fillStyle=i===levels.length-1?"#75ff5b":"#00e5ff";
    x.beginPath(); x.arc(xx,yy,6,0,Math.PI*2); x.fill();
    x.fillStyle="#dff8ff"; x.font="10px monospace";
    x.fillText(p[0],xx-32,h-18);
    x.fillText(n(p[1],2)+" dBm",xx-32,yy-12);
  });

  x.strokeStyle=c.linPass?"#75ff5b":"#ffd84d";
  x.beginPath();
  for(let i=0;i<w;i++){
    const y = h/2 + Math.sin((i+tick)/35)*18 + Math.sin((i+tick)/9)*4 + c.attResidual*600;
    if(i===0)x.moveTo(i,y); else x.lineTo(i,y);
  }
  x.stroke();
}

function drawPhaseNoise(){
  const canvas = $("pnCanvas"), r = canvas.getBoundingClientRect(), dpr = devicePixelRatio || 1;
  canvas.width = r.width*dpr; canvas.height = r.height*dpr;
  const x = canvas.getContext("2d"); x.setTransform(dpr,0,0,dpr,0,0);
  const w = r.width, h = r.height, c = calc();

  x.fillStyle="#020b12"; x.fillRect(0,0,w,h);
  x.strokeStyle="rgba(0,210,255,.16)";
  for(let i=0;i<w;i+=50){x.beginPath();x.moveTo(i,0);x.lineTo(i,h);x.stroke();}
  for(let j=0;j<h;j+=34){x.beginPath();x.moveTo(0,j);x.lineTo(w,j);x.stroke();}

  function trace(color, offset){
    x.strokeStyle=color; x.lineWidth=1.5; x.beginPath();
    for(let i=0;i<w;i++){
      const f = Math.max(1,i);
      const y = 30 + Math.log10(f+1)*38 + offset + Math.sin((i+tick)/31)*3 + Math.random()*2;
      if(i===0)x.moveTo(i,y); else x.lineTo(i,y);
    }
    x.stroke();
  }
  trace("#ffd400",0);
  trace("#75ff5b",-c.xcorrImprove*2.5);

  x.fillStyle="#83cfff"; x.font="10px monospace";
  x.fillText("Phase noise raw",16,22);
  x.fillStyle="#75ff5b";
  x.fillText("Cross-correlation improved",16,40);
}
